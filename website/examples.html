Examples

Whole programs

12 programs that use cartography, from a first map to terrain analysis. Each one passes tessel check, and the ones that draw or print were run to make what's shown here.

To run one: make a project with tessel new maps, then tessel add cartography in it. Put the program in main.tsl and run tessel run .. Examples that call geodesy or viewport themselves also need tessel add geodesy or tessel add viewport.

Data: countries.geojson, rivers.geojson and places.geojson are in the library's examples/data (from Natural Earth); dem.tif is in tests/data/raster. For the others, use files of your own.

In the repository: each program is in examples/gallery, in a folder of its own.

A first map

OpenStreetMap in a window, with two towns made in code as red stars with their names. MapView pans, zooms, downloads tiles and draws the scale bar and north arrow.

first-map.tslan app · 25 lines
import cartography

// A layer of points made in code: two towns, as red stars with their names.
fn towns() -> cartography.Layer {
    var t = cartography.Layer(name: "Towns", geomKind: .point)
    t.geoms = [cartography.pointGeometry(121.0, 14.6), cartography.pointGeometry(120.6, 16.4)]
    t.columns = [cartography.textColumn(name: "name", values: ["Manila", "Baguio"])]
    t.style.symbol = cartography.Symbol(fill: cartography.rgb(220, 40, 40), size: 12, marker: .star)
    t.labels = cartography.LabelStyle(enabled: true, field: "name", bold: true)
    t
}

fn startingMap() -> cartography.Map {
    var m = cartography.Map()                  // Web Mercator
    m.add(cartography.openStreetMap())
    m.add(towns())                             // also places the map on them
    m.centerOn(lon: 121.0, lat: 15.5, scale: 2000000)
    m
}

app FirstMap(title: "First map", width: 800, height: 600) {
    state m = startingMap()

    cartography.MapView(map: m, northArrow: true)
}

Find places by name

A search box filters a layer: features whose names don't match go into hidden, which leaves them out of the drawing and of clicks. Clicking a place shows its name and country. On CARTO Positron tiles.

hiddenrefreshfitLayerattributestileSource
find-places.tslan app · 51 lines
import cartography

fn startingMap() -> cartography.Map {
    var m = cartography.Map()
    m.add(cartography.tileLayer(cartography.tileSource(named: "CARTO Positron") ?? cartography.tileSources()[0]))
    if let layers = cartography.readLayers("places.geojson").value() {
        var places = layers[0]
        places.style.symbol = cartography.Symbol(fill: cartography.rgb(21, 128, 61), size: 7)
        places.labels = cartography.LabelStyle(enabled: true, field: "NAME", size: 10, background: true)
        m.add(places)
    }
    m
}

app FindPlaces(title: "Find places", width: 900, height: 620) {
    state m = startingMap()
    state query = ""
    state found = ""

    VStack(spacing: 0) {
        HStack {
            TextField("Find a place", text: query).onSubmit { search() }
            Button("Find") { search() }
            Button("Show all") {
                query = ""
                search()
            }
            Text(found).color(.secondary)
        }
        .padding(8)
        cartography.MapView(map: m) { hit in
            if let h = hit {
                found = m.attributes(h).filter { p in p.0 == "NAME" || p.0 == "ADM0NAME" }
                    .map { p in p.1 }.joined(separator: ", ")
            }
        }
    }

    // Hides the places whose names don't have the query in them, then
    // shows the ones that are left.
    fn search() {
        if let i = m.layerIndex(named: "places") {
            let q = query.trim().lowercase()
            let names = m.layers[i].columnIndex("NAME") ?? 0
            m.layers[i].hidden = m.layers[i].columns[names].texts.map { n in q != "" && !n.lowercase().contains(q) }
            m.refresh(i)
            found = "{m.layers[i].shownCount()} places"
            m.fitLayer(i)
        }
    }
}

A GPX track over a choice of basemaps

Reads a GPX file's track and waypoints. The track is a cased line with arrows along it showing which way it goes; the waypoints are labeled to their right. A menu swaps the basemap at the bottom of the map.

// A GPX track over a choice of basemaps, its direction shown by arrows
// along it, with the waypoints labeled.
fn startingMap() -> cartography.Map {
    var m = cartography.Map()
    m.add(cartography.tileLayer(cartography.tileSource(named: "OpenTopoMap") ?? cartography.tileSources()[0]))
    match cartography.readLayers("hike.gpx") {
        .failure(e) -> print(e.message)
        .ok(layers) -> {
            for read in layers {
                var layer = read
                if layer.geomKind == .line {
                    layer.style.symbol = cartography.Symbol(stroke: cartography.rgb(200, 30, 90), strokeWidth: 3,

                m.add(layer)
            }
        }
    }
    m
}

The same OpenStreetMap tiles, warped into UTM zone 51N. The scale is measured on the ground, the mouse shows eastings and northings, and buttons redraw the map in other systems without losing its place.

import cartography
import geodesy

// A map drawn in UTM zone 51N rather than Web Mercator: OpenStreetMap's
// tiles are warped to fit, the scale is measured on the ground, and the
// mouse shows eastings and northings.
fn startingMap() -> cartography.Map {
    var m = cartography.Map(crs: geodesy.epsg(32651) ?? cartography.webMercator())
    m.add(cartography.openStreetMap())
    m.credit = "UTM 51N"
    m.centerOn(lon: 121.0, lat: 14.6, scale: 50000)
    m
}

app Utm(title: "UTM", width: 900, height: 640) {
    state m = startingMap()
    state xy = ""

    VStack(spacing: 0) {
        cartography.MapView(map: m, imperial: false, keys: true, onMouse: { lon, lat in
            let p = cartography.transformCoords(from: cartography.lonLat(), to: m.crs, coords: [lon, lat])
            xy = "E {cartography.groupedNumber(p[0])}  N {cartography.groupedNumber(p[1])}"
        })
        HStack {
            Button("UTM 51N") { m.setCrs(geodesy.epsg(32651) ?? cartography.webMercator()) }
            Button("Web Mercator") { m.setCrs(cartography.webMercator()) }
            Button("Degrees") { m.setCrs(cartography.lonLat()) }
            Spacer()
            Text(xy)
            Text("1:{cartography.groupedNumber(m.scaleDenominator())}").color(.secondary)
        }
        .font(size: 12)
        .padding(8)
    }
}

A map on your own Canvas

Everything MapView does by hand: resizing the viewport, passing it pointer events, downloading the tiles each new view needs, and clicking to select. A button selects the features in view with featuresIn.

Canvasvp.handlefetchTilesfeaturesIndrawNorthArrow
own-canvas.tslan app · 80 lines
import cartography
import viewport

// A map on a Canvas of your own rather than MapView: it pans and zooms
// through the map's viewport, downloads the tiles each new view needs, and
// counts the features inside a box drawn with Shift and a drag.
fn startingMap() -> cartography.Map {
    var m = cartography.Map()
    m.add(cartography.openStreetMap())
    if let layers = cartography.readLayers("buildings.gpkg").value() {
        for layer in layers {
            m.add(layer)
        }
    }
    m.vp.smooth = false                     // no animation, so no timer needed
    m
}

app OwnCanvas(title: "Own canvas", width: 900, height: 640) {
    state m = startingMap()
    state redraw = 0
    state status = "Drag to pan, wheel to zoom"

    VStack(spacing: 0) {
        Canvas { g in
            let _r = redraw                 // draw again when a tile comes in
            m.draw(g)
            viewport.drawZoomBox(g, m.vp)
            cartography.drawMapScaleBar(g, m, x: 14, y: g.height - 22)
            cartography.drawNorthArrow(g, m, x: g.width - 30, y: 34)
            let _size = cartography.drawLegend(g, m, x: 10, y: 10, maxEntries: 12)
        }
        .onResize { w, h in
            m.vp.resize(w, h)
            fetch()
        }
        .onPointer { e in
            if m.vp.handle(e) {
                fetch()
            }
            if e.kind == .up && e.button == .left && e.clicks == 1 {
                if let h = m.featureAt(e.x, e.y) {
                    m.select(layer: h.layer, rows: [h.row])
                    status = "{m.layers[h.layer].name}, row {h.row}"
                } else {
                    m.clearSelection()
                }
            }
        }
        HStack {
            Text(status)
            Spacer()
            Button("Count in view") { countInView() }
        }
        .padding(8)
    }

    // Starts downloading the tiles the view needs and isn't showing yet.
    fn fetch() {
        let started = cartography.fetchTiles(m, userAgent: "OwnCanvas/1.0", arrived: { path, ok in
            m.loading.remove(path)
            redraw += 1
        }) { paths in
            if cartography.warpTileLayers(m) {     // only for maps not in Web Mercator
                redraw += 1
            }
        }
        for path in started {
            m.loading.insert(path)
        }
    }

    // How many features of the top layer are in the view.
    fn countInView() {
        let top = m.layers.count - 1
        let rows = m.featuresIn(layer: top, 0, 0, m.vp.width, m.vp.height)
        m.select(layer: top, rows: rows)
        status = "{rows.count} of {m.layers[top].featureCount()} in view"
    }
}

The world in Equal Earth

Countries colored by continent, rivers, and only the cities of eight million people or more, in the Equal Earth projection. Saved as a PNG with a legend and scale bar, without a window.

categorizedStylehiddenrenderImagedrawLegendEPSG:8857
world.tsla program · 49 lines
import cartography
import geodesy

fn layerOf(_ path: String) -> cartography.Layer {
    match cartography.readLayers(path) {
        .ok(layers) -> layers[0]
        .failure(e) -> {
            print(e.message)
            exit(1)
            cartography.Layer()
        }
    }
}

fn main() {
    // Equal Earth (EPSG:8857), on a pale blue sea.
    var m = cartography.Map(crs: geodesy.epsg(8857) ?? cartography.lonLat(), background: .rgb(214, 232, 245))

    // Countries: a color for each continent, with white borders.
    var countries = layerOf("countries.geojson")
    countries.style = cartography.categorizedStyle(layer: countries, field: "CONTINENT")
    for i in 0..countries.style.classes.count {
        countries.style.classes[i].symbol.stroke = cartography.rgb(255, 255, 255)
        countries.style.classes[i].symbol.strokeWidth = 0.5
    }
    m.add(countries)

    var rivers = layerOf("rivers.geojson")
    rivers.style.symbol = cartography.Symbol(stroke: cartography.rgb(70, 130, 200), strokeWidth: 1.0)
    m.add(rivers)

    // Only the cities of eight million people or more.
    var places = layerOf("places.geojson")
    let pop = places.columnIndex("POP_MAX") ?? 0
    places.hidden = places.columns[pop].nums.map { p in p < 8000000.0 }
    places.style.symbol = cartography.Symbol(fill: cartography.rgb(40, 40, 40), stroke: cartography.rgb(255, 255, 255), size: 5)
    places.labels = cartography.LabelStyle(enabled: true, field: "NAME", size: 10)
    m.add(places)

    m.vp.resize(1000, 520)
    m.vp.smooth = false
    m.fitAll()
    let ok = renderImage(path: "world.png", width: 1000, height: 520, scale: 2) { g in
        m.draw(g)
        let _size = cartography.drawLegend(g, m, x: 12, y: 12)
        cartography.drawMapScaleBar(g, m, x: 16, y: 504)
    }
    print("world.png: {ok}")
}

Population by country

A graduated style: five classes of population by natural breaks, colored yellow to red, each class labeled in millions for the legend.

graduatedStylenaturalBreaksStyleClassformatNumber
choropleth.tsla program · 42 lines
import cartography
import geodesy

fn millions(_ x: Float) -> String {
    cartography.formatNumber(round(x / 100000.0) / 10.0, decimals: 1)
}

fn main() {
    var layers: [cartography.Layer] = []
    match cartography.readLayers("countries.geojson") {
        .ok(read) -> { layers = read }
        .failure(e) -> {
            print(e.message)
            return
        }
    }
    var m = cartography.Map(crs: geodesy.epsg(8857) ?? cartography.lonLat(), background: .rgb(245, 245, 245))

    // Five classes of population by natural breaks (Jenks), yellow to red,
    // each labeled in millions.
    var countries = layers[0]
    countries.name = "Population"
    countries.style = cartography.graduatedStyle(layer: countries, field: "POP_EST",
        mode: .naturalBreaks, count: 5, ramp: "YlOrRd")
    for i in 0..countries.style.classes.count {
        let c = countries.style.classes[i]
        countries.style.classes[i].label = "{millions(c.lower)} – {millions(c.upper)} M"
        countries.style.classes[i].symbol.stroke = cartography.rgb(120, 120, 120)
        countries.style.classes[i].symbol.strokeWidth = 0.4
    }
    m.add(countries)

    m.vp.resize(1000, 520)
    m.vp.smooth = false
    m.fitAll()
    let ok = renderImage(path: "population.png", width: 1000, height: 520, scale: 2) { g in
        m.draw(g)
        let _size = cartography.drawLegend(g, m, x: 12, y: 12)
        cartography.drawMapScaleBar(g, m, x: 16, y: 504)
    }
    print("population.png: {ok}")
}

Terrain: hillshade and contours

A DEM colored along the Terrain ramp over its own shaded relief, with contour lines every 25 meters labeled along their curves. The contours are also saved as a GeoPackage.

readRasterhillshaderampRasterStylecontourscurved
terrain.tsla program · 46 lines
import cartography

fn main() {
    var m = cartography.Map()
    match cartography.readRaster("dem.tif") {
        .failure(e) -> print(e.message)
        .ok(read) -> {
            var dem = read
            dem.name = "Elevation (m)"

            // Shaded relief underneath, with the land drawn twice as tall.
            var hills = cartography.hillshade(dem, exaggeration: 2.0)
            hills.rasterStyle.look = .gray
            hills.rasterStyle.low = 0
            hills.rasterStyle.high = 255
            m.add(hills)

            // The heights in color, three quarters opaque so the hills show.
            dem.rasterStyle = cartography.rampRasterStyle(dem, ramp: "Terrain")
            dem.opacity = 0.75
            m.add(dem)

            // Contours every 25 m, labeled along their lines.
            var lines = cartography.contours(dem, interval: 25)
            lines.style.symbol = cartography.Symbol(stroke: cartography.Rgba(r: 60, g: 40, b: 20, a: 0.6), strokeWidth: 0.8)
            lines.labels = cartography.LabelStyle(enabled: true, field: "elev", size: 9,
                color: cartography.rgb(80, 50, 20), curved: true)
            m.add(lines)

            m.vp.resize(700, 520)
            m.vp.smooth = false
            m.fitAll()
            let ok = renderImage(path: "terrain.png", width: 700, height: 520, scale: 2) { g in
                m.draw(g)
                let _size = cartography.drawLegend(g, m, x: 12, y: 12)
                cartography.drawMapScaleBar(g, m, x: 16, y: 504)
            }
            print("terrain.png: {ok}")

            // The contours as a GeoPackage, for QGIS.
            if let e = cartography.writeLayer(lines, "contours.gpkg") {
                print(e.message)
            }
        }
    }
}

import cartography

// Rain gauges from a CSV file (with lon and lat columns), colored by their
// kind and sized by their yearly rainfall, saved as a picture over
// OpenStreetMap and as a GeoJSON file.
fn main() {
    let text = readFile("gauges.csv") ?? ""
    let info = cartography.inspectCSV(text)
    print("{info.rows} rows; x = {info.xField}, y = {info.yField}; fields: {info.fields.joined(separator: ", ")}")

    match cartography.readCSV(text, name: "Rain gauges") {
        .failure(e) -> print(e.message)
        .ok(read) -> {
            var gauges = read
            gauges.style = cartography.categorizedStyle(layer: gauges, field: "kind")
            gauges.style.sizeField = "rain_mm"
            gauges.style.sizeFrom = 6
            gauges.style.sizeTo = 22
            // Names only from 1:1,000,000 in.

            m.add(cartography.openStreetMap())
            m.add(gauges)
            m.vp.resize(900, 600)
            m.vp.smooth = false
            m.fitAll()

            // Tiles first, then the picture.
            let _started = cartography.fetchTiles(m) { paths in
                let ok = renderImage(path: "gauges.png", width: 900, height: 600, scale: 2) { g in
                    m.draw(g)
                    let _size = cartography.drawLegend(g, m, x: 12, y: 12)
                    cartography.drawMapScaleBar(g, m, x: 16, y: 584)
                }
                print("gauges.png: {ok}")

                print(e.message)
            }
        }
    }

import cartography

// Reads a shapefile (in whatever system its .prj says) and writes it as
// GeoJSON, KML, a GeoPackage and CSV, saying what it found.
fn main() {
    match cartography.readLayers("lots.shp") {
        .failure(e) -> print("lots.shp: {e.message}")
        .ok(layers) -> {
            for layer in layers {
                print("{layer.name}: {layer.featureCount()} features")
                print("  fields: {layer.fieldNames().joined(separator: ", ")}")
                let b = layer.extent()
                print("  extent: {b.minX}, {b.minY} – {b.maxX}, {b.maxY}")
            }
            let lots = layers[0]
            for path in ["lots.geojson", "lots.kml", "lots.csv"] {
                if let e = cartography.writeLayer(lots, path) {
                    print("{path}: {e.message}")
                } else {
                    print("wrote {path}")
                }
            }
            // A GeoPackage can hold several layers.
            if let e = cartography.writeGeoPackage(layers, "lots.gpkg") {
                print("lots.gpkg: {e.message}")
            }
        }
    }
}

A layer made in code

Survey points in UTM built feature by feature with addFeature, one missing a value. Printed as WKT and as GeoJSON (always in longitude and latitude), and saved as a shapefile and a GeoPackage.

import cartography
import geodesy

// A layer of survey points made in code, in UTM zone 51N, saved as a
// shapefile and a GeoPackage, and printed as WKT and GeoJSON.
fn main() {
    let utm = geodesy.epsg(32651) ?? cartography.lonLat()
    var pts = cartography.newVectorLayer(name: "Survey", geomKind: .point, crs: utm)
    pts.addColumn(name: "id", kind: .int)
    pts.addColumn(name: "label", kind: .text)
    pts.addColumn(name: "elev", kind: .float)

    pts.addFeature(cartography.pointGeometry(500120.0, 1615230.0),
        values: [.num(value: 1), .text(value: "BM-1"), .num(value: 12.48)])
    pts.addFeature(cartography.pointGeometry(500410.5, 1615602.2),
        values: [.num(value: 2), .text(value: "BM-2"), .num(value: 14.02)])
    pts.addFeature(cartography.pointGeometry(500733.1, 1615188.9),
        values: [.num(value: 3), .text(value: "BM-3")])     // no elevation yet

    for i in 0..pts.featureCount() {
        let row = pts.row(i).map { v in v.display() }
        print("{cartography.wkt(pts.geoms[i], decimals: 2)}  {row.joined(separator: " | ")}")
    }

    if let e = cartography.writeLayer(pts, "survey.shp") {
        print(e.message)
    }
    if let e = cartography.writeLayer(pts, "survey.gpkg") {
        print(e.message)
    }
    // GeoJSON is always in longitude and latitude.
    print(cartography.geoJSON(pts, decimals: 6))
}

Numbers from a DEM

A DEM's statistics and histogram, a profile across it, the mean height in each zone of a GeoPackage, and its slope saved as an LZW GeoTIFF.

raster-stats.tsla program · 55 lines
import cartography

// Numbers from a DEM: its statistics, a histogram, a profile along a
// line, the mean height in each zone of a GeoPackage, and its slope saved
// as a GeoTIFF.
fn main() {
    let dem = match cartography.readRaster("dem.tif") {
        .ok(layer) -> layer
        .failure(e) -> {
            print(e.message)
            return
        }
    }
    let r = dem.raster
    let s = cartography.rasterStatistics(r)
    print("{r.width} × {r.height} cells, {s.count} with a value")
    print("height {cartography.formatNumber(s.min, decimals: 1)} to {cartography.formatNumber(s.max, decimals: 1)} m, mean {cartography.formatNumber(s.mean, decimals: 1)} m")

    // Ten equal steps from the lowest to the highest.
    let bins = cartography.rasterHistogram(r, bins: 10)
    let step = (s.max - s.min) / 10.0
    let most = bins.max() ?? 1
    for i in 0..bins.count {
        var bar = ""
        for _ in 0..(bins[i] * 40 / most) {
            bar += "#"
        }
        print("  {cartography.formatNumber(s.min + step * Float(i), decimals: 0)} m  {bar}")
    }

    // A profile across the raster, 50 points from west to east.
    let b = r.box
    var points: [Float] = []
    for i in 0..50 {
        points.append(b.minX + b.width() * Float(i) / 49.0)
        points.append(b.centerY())
    }
    let profile = cartography.sampleRaster(r, points: points)
    print("profile: {profile.map { z in cartography.formatNumber(z, decimals: 0) }.joined(separator: " ")}")

    // The mean height in each zone (the zones must be in the DEM's system).
    if let zones = cartography.readLayers("zones.gpkg").value() {
        let z = zones[0]
        let shapes = z.geoms.map { g in cartography.reprojected(g, from: z.crs, to: dem.crs) }
        let stats = cartography.zonalStatistics(r, zones: shapes)
        for i in 0..stats.count {
            print("zone {i}: {stats[i].count} cells, mean {cartography.formatNumber(stats[i].mean, decimals: 1)} m")
        }
    }

    let steep = cartography.slope(dem)
    if let e = cartography.writeGeoTIFF(steep, "slope.tif", compression: "lzw") {
        print(e.message)
    }
}
